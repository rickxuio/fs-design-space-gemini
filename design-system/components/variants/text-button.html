⚠️ 硬规则：卡片内 Learn more 固定变体
描述原文
Card / IconCard 的 Learn more 使用对应宿主契约规定的黑色常规文字按钮。独立TextButton只开放当前aiContract的真实变体、尺寸、官方右箭头和原有显示开关；不强套来源左图标或折叠控件。既有宿主的Chevron与展开态仅在该宿主契约范围保留，不作为独立TextButton新增能力。默认和Hover不能混用，不把下划线常驻Default。
AI 解析（写入契约 JSON 的字段）
relations.commonlyEmbeddedIn[0] = {
  component: "Card",
  usage: "vertical/horizontal 卡片的 Learn more 链接，固定 ds-text-btn--inline ds-text-btn--black ds-text-btn--m，父级依赖 margin-top:auto 居底"
}

relations.commonlyEmbeddedIn[1] = {
  component: "IconCard",
  usage: "单事件卡片的 Learn more 链接，同 Card 用法，位于 ds-icon-card__link 行内"
}
